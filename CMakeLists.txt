cmake_minimum_required(VERSION 3.22)
project(agbplay VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(ENABLE_ADDRESS_SANITIZER "Enable Address Sanitizer and Undefined Behavior Sanitizer" OFF)
option(AGBPLAY_FETCH_DEPENDENCIES "Fetch pinned portable dependencies (Android)" ${ANDROID})
option(AGBPLAY_BUILD_GUI "Build Qt Widgets application" ON)
if(ANDROID)
    set(AGBPLAY_DESKTOP_DEFAULT OFF)
else()
    set(AGBPLAY_DESKTOP_DEFAULT ON)
endif()
option(AGBPLAY_BUILD_CLI "Build command line exporter" ${AGBPLAY_DESKTOP_DEFAULT})
option(AGBPLAY_BUILD_NC "Build terminal player" ${AGBPLAY_DESKTOP_DEFAULT})
option(AGBPLAY_QT_AUDIO "Use Qt Multimedia instead of PortAudio" ${ANDROID})
option(AGBPLAY_ENABLE_AVX2 "Build x86 AVX2 resamplers" ON)
include(CTest)
if(ENABLE_ADDRESS_SANITIZER)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
if(AGBPLAY_FETCH_DEPENDENCIES)
    include(cmake/Dependencies.cmake)
endif()
add_subdirectory(src/agbplay)
if(AGBPLAY_BUILD_GUI OR AGBPLAY_BUILD_NC OR BUILD_TESTING)
    add_subdirectory(src/agbplay-util)
endif()
if(AGBPLAY_BUILD_NC AND NOT WIN32 AND NOT ANDROID)
    add_subdirectory(src/agbplay-nc)
endif()
if(AGBPLAY_BUILD_GUI)
    add_subdirectory(src/agbplay-gui)
endif()
if(AGBPLAY_BUILD_CLI)
    add_subdirectory(src/agbplay-cli)
endif()
if(BUILD_TESTING)
    add_subdirectory(src/tests)
endif()
